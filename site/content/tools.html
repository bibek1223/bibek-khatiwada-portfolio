Semantic Similarity & Vector Scoring Engine

A peek inside the Python spaCy + sentence-transformers (all-MiniLM-L6-v2) cosine distance pipeline used to audit header hierarchy against user intent.

semantic_flow_checker.py 
from sentence_transformers import SentenceTransformer, util
import spacy

# Load lightweight transformer model and spaCy parser
nlp = spacy.load("en_core_web_sm")
model = SentenceTransformer("all-MiniLM-L6-v2")

def analyze_semantic_flow(query: str, headers: list[str]):
    query_emb = model.encode(query, convert_to_tensor=True)
    header_embs = model.encode(headers, convert_to_tensor=True)
    
    # Calculate cosine similarity matrix
    scores = util.cos_sim(query_emb, header_embs)[0]
    
    return [
        {"header": h, "similarity_score": round(float(score), 4)}
        for h, score in zip(headers, scores)
    ]

# Example Execution
results = analyze_semantic_flow(
    query="How to recover medical YMYL website from Google core update",
    headers=[
        "Diagnosing algorithm drops by update date",
        "Content pruning and directory consolidation mechanics",
        "Best pricing strategies for retail products"  # Off-topic drift
    ]
)
print(results)
# Output: 0.8421 (High), 0.7914 (High), 0.1102 (Off-topic Drift Flagged)
